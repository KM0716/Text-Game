```javascript
function applyEffects(effects) {
  try {
    if (!effects) return;
    const s = window.gst ? window.gst() : null;
    const c = s ? window.gclk() : null;
    // 状态变更 hp/hunger/thirst/fatigue/spirit/infection/bodyTemp
    ['hp','hunger','thirst','fatigue','spirit','infection'].forEach(k => {
      if (effects[k] != null && s) { s[k] = Math.max(0, Math.min(100, (s[k] ?? 50) + effects[k])); }
    });
    if (effects.bodyTemp != null && s) { s.bodyTemp = Math.max(30, Math.min(45, (s.bodyTemp ?? 37) + effects.bodyTemp)); }
    // 加物品
    if (effects.invAdd && Array.isArray(effects.invAdd) && typeof window.invAdd === 'function') {
      effects.invAdd.forEach(it => { try { window.invAdd(it, 1); } catch(_) {} });
    }
    // 减物品
    if (effects.invRemove && Array.isArray(effects.invRemove) && typeof window.invRemove === 'function') {
      effects.invRemove.forEach(it => { try { window.invRemove(it, 1); } catch(_) {} });
    }
    // 解锁线索
    if (effects.clue && s) {
      if (!s.clues) s.clues = [];
      const txt = effects.clue;
      const exist = s.clues.some(c => (typeof c === 'string' ? c === txt : (c && c.text === txt)));
      if (!exist) s.clues.push({ text: txt, priority: 2, id: 'clue_evt_' + Date.now() + Math.random().toString(36).slice(2,6), time: (window.gclk().day||1)+'日' });
      if (typeof window.sst === 'function') window.sst(s);
    }
    // 解锁地图
    if (effects.mapUnlock && s) {
      if (!s.mapUnlock) s.mapUnlock = [];
      if (!s.mapUnlock.includes(effects.mapUnlock)) s.mapUnlock.push(effects.mapUnlock);
      if (typeof window.sst === 'function') window.sst(s);
    }
    // 改变天气/气温
    if ((effects.weather || effects.temp !== undefined) && c) {
      if (effects.weather) c.weather = effects.weather;
      if (effects.temp !== undefined) c.temp = effects.temp;
      if (typeof window.sclk === 'function') window.sclk(c);
    }
    // 改变位置
    if (effects.location && s) {
      s.location = effects.location;
      if (typeof window.sst === 'function') window.sst(s);
    }
    if (typeof window.upui === 'function') { try { window.upui(); } catch(_) {} }
  } catch(e) { console.error('applyEffects error:', e); }
}
```

### 2.1 新增 checkAndRemoveCompletedClues 函数
**位置**：`main.js`，紧接在 `generateNeutralChoices` 之后、`hin` 之前（约 3362 行附近）

```javascript
function checkAndRemoveCompletedClues(storyText) {
  try {
    const s = gst();
    if (!s || !s.clues || !s.clues.length || !storyText) return;
    const beforeCount = s.clues.length;
    const suffixes = ['已解决','已完成','已证实','已处理','已找到','已确认','不需要了','没用了','结束了','搞定了'];
    // 模式1：线索文本 + 后缀词
    s.clues = s.clues.filter(c => {
      const txt = (typeof c === 'string') ? c : (c && c.text ? c.text : '');
      if (!txt) return true;
      // 直接包含 线索+后缀
      for (const suf of suffixes) {
        if (storyText.includes(txt + suf)) return false;
      }
      // 模式2：销毁/删除/丢弃线索相关模式
      const destroyPatterns = [
        new RegExp('销毁了.*' + txt.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '.*线索'),
        new RegExp('删除了.*' + txt.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '.*线索'),
        new RegExp('丢弃了.*' + txt.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '.*(线索|便签)'),
        /销毁了(有关|关于)?.*(线索|便签)/,
        /删除了(有关|关于)?.*(线索|便签)/,
        /丢弃了(有关|关于)?.*(线索|便签)/
      ];
      for (const pat of destroyPatterns) {
        if (pat.test(storyText)) return false;
      }
      return true;
    });
    const removed = beforeCount - s.clues.length;
    if (removed > 0) {
      sst(s);
      try { renderClueSidebar(); } catch(_) {}
      try { snotify('info', '线索完成', '已清理 ' + removed + ' 条完成线索'); } catch(_) {}
    }
  } catch(e) { console.warn('checkAndRemoveCompletedClues err:', e); }
}
```

```javascript
          x = x.replace(/^\s*[\-—–•●*·]+\s*/, '');
          x = x.replace(/^["""''`]+|["""''`]+$/g, '').trim();
          // 去掉非法标签符号
          x = x.replace(/\[[^\]]*\]/g, '').trim();
          // 截断长度
          if (x.length > 14) x = x.slice(0, 13) + '…';
          if (!x) continue;
          const k = x.toLowerCase();
          if (!seen.has(k)) { seen.add(k); out.push(x); }
          if (out.length >= 4) break;
        }
        return out;
      } catch(e) { console.warn('validateChoices err:', e); return []; }
    },

    validateTags(tagList) {
      try {
        if (!Array.isArray(tagList)) return [];
        return tagList.filter(tg => {
          if (typeof tg !== 'string') return false;
          const norm = tg.replace(/：/g, ':').replace(/:\s+/g, ':');
          for (const [prefix] of ALLOWED_TAGS) {
            if (norm.startsWith(prefix)) return true;
          }
          return false;
        });
      } catch(e) { console.warn('validateTags err:', e); return []; }
    },

    validateOutput(text) {
      try {
        let result = { narrative: '', choices: [], tags: [], raw: text || '' };
        if (!text || typeof text !== 'string') return result;
        // 提取剧情正文
        const narrMatch = text.match(/【剧情正文】([\s\S]*?)(?=【选项】|【标签】|$)/i);
        if (narrMatch) result.narrative = narrMatch[1].trim();
        // 提取选项
        const choicesMatch = text.match(/【选项】([\s\S]*?)(?=【标签】|$)/i);
        if (choicesMatch) {
          const lines = choicesMatch[1].split(/\n/).map(l => l.trim()).filter(Boolean);
          result.choices = VALIDATORS.validateChoices(lines);
        }
        // 提取标签
        const tagsMatch = text.match(/【标签】([\s\S]*)$/i);
        if (tagsMatch) {
          const tags = (tagsMatch[1].match(/\[[^\]]+\]/g) || []).map(t => t.slice(1, -1));
          result.tags = VALIDATORS.validateTags(tags);
        }
        // 兜底补齐
        if (!result.narrative) {
          // 尝试直接把整段没有【】的当叙事
          result.narrative = text.replace(/【[^】]*】[\s\S]*?$/g, '').trim();
        }
        if (result.choices.length < 3) {
          const fallback = FALLBACKS.fallbackChoices({});
          while (result.choices.length < 3 && fallback.length) {
            result.choices.push(fallback.shift());
          }
        }
        return result;
      } catch(e) { console.warn('validateOutput err:', e); return { narrative: text||'', choices: [], tags: [] }; }
    }
  };

  // d) 兜底函数
  const FALLBACKS = {
    fallbackChoices(ctx) {
      try {
        const out = ['观察周围环境', '检查随身装备', '原地警戒片刻'];
        if (!ctx) return out;
        const s = (typeof window.gst === 'function') ? window.gst() : null;
        if (s) {
          if ((s.hunger ?? 50) < 40) out.push('寻找食物');
          if ((s.thirst ?? 50) < 40) out.push('寻找水源');
          if ((s.fatigue ?? 30) > 70) out.push('找地方休息');
          if ((s.hp ?? 100) < 50) out.push('包扎伤口');
        }
        const clk = (typeof window.gclk === 'function') ? window.gclk() : null;
        if (clk) {
          const h = Math.floor((clk.elapsedSec || 0) / 3600) % 24;
          if (h >= 20 || h < 6) out.push('寻找夜间藏身点');
        }
        return out.slice(0, 4);
      } catch(_) { return ['观察周围环境', '检查随身装备', '原地警戒片刻']; }
    },

    fallbackNarrative(ctx) {
      try {
        const s = (typeof window.gst === 'function') ? window.gst() : null;
        const loc = (s && s.location) ? s.location : '这片区域';
        const clk = (typeof window.gclk === 'function') ? window.gclk() : null;
        const phase = clk ? window.dayPhase ? window.dayPhase(clk.elapsedSec) : '' : '';
        const sentences = [
          `你环顾四周，${phase ? phase + '的' : ''}${loc}弥漫着末日特有的寂静。`,
          `远处隐约传来细微的动静，你停下脚步，警觉地扫视着周围的废墟。`,
          `空气中弥漫着尘埃和腐朽的气味，你深吸一口气，继续保持警惕。`
        ];
        return sentences.join('');
      } catch(_) {
        return '你环顾四周，末日的寂静笼罩着这片区域，远处传来若有若无的声响，让你不由自主地握紧了手中的武器。';
      }
    }
  };

  // e) 标签辅助函数
  function buildTagHelpers(state) {
    const h = {};
    h.addItem = (name) => '[物品:+' + name + ']';
    h.remItem = (name) => '[物品:-' + name + ']';
    h.addClue = (t) => '[线索:' + t + ']';
    h.mapUnlock = (name) => '[地图:解锁-' + name + ']';
    h.setWeather = (w) => '[天气:' + w + ']';
    h.setTemp = (v) => '[气温:' + v + ']';
    h.setHp = (v) => '[血量:' + Math.max(0, Math.min(100, v)) + ']';
    h.setHunger = (v) => '[饱腹:' + Math.max(0, Math.min(100, v)) + ']';
    h.setThirst = (v) => '[口渴:' + Math.max(0, Math.min(100, v)) + ']';
    h.setFatigue = (v) => '[疲劳:' + Math.max(0, Math.min(100, v)) + ']';
    h.setSpirit = (v) => '[精神:' + Math.max(0, Math.min(100, v)) + ']';
    h.setLoc = (name) => '[地点:' + name + ']';
    h.advTime = (v) => '[时间:+' + v + 'h]';
    h.addTrait = (name) => '[特质:+' + name + ']';
    h.remTrait = (name) => '[特质:-' + name + ']';
    h.addSkill = (name) => '[技能:+' + name + ']';
    h.remSkill = (name) => '[技能:-' + name + ']';
    return h;
  }

  // 暴露全局
  window.AI_FRAMEWORK = {
    OUTPUT_TEMPLATES,
    ALLOWED_TAGS,
    VALIDATORS,
    FALLBACKS,
    buildTagHelpers
  };
})();
```
